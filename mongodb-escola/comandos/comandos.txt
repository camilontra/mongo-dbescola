# Consultas realizadas

# Atualização
Edição manual de idade de um aluno (Compass)

# Adição em array
Adição de nova nota no campo "notas" (Compass)

# Remoção
Exclusão de um aluno (Compass)

# Aggregation 1 - Média de notas por aluno
{
  $project: {
    nome: 1,
    media: { $avg: "$notas" }
  }
}

# Aggregation 2 - Quantidade de alunos por curso
{
  $group: {
    _id: "$curso",
    quantidade: { $sum: 1 }
  }
}
